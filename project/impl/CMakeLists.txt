cmake_minimum_required(VERSION 3.20)
project(game_library CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
add_library(game_library STATIC
    src/user.cpp src/developer.cpp src/game.cpp src/review.cpp
    src/repository.cpp src/review_comment.cpp src/repo_comment.cpp src/game_page.cpp)
target_include_directories(game_library PUBLIC include)
add_executable(demo src/demo.cpp)
target_link_libraries(demo PRIVATE game_library)
add_executable(scenario_user scenarios/user.cpp)
target_link_libraries(scenario_user PRIVATE game_library)
foreach(target game_library demo scenario_user)
    if(MSVC)
        target_compile_options(${target} PRIVATE /W4)
    else()
        target_compile_options(${target} PRIVATE -Wall -Wextra)
    endif()
endforeach()
